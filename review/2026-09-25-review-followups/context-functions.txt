539: function decryptWallet( keyset ){
540:     if(!keyset) keyset = FW.WALLET_ENCKEY;
541:     if(!keyset) return false;
542:     var v = readVault();
543:     if(!v || !v.wallet) return false;
544:     var wallet   = WalletCrypto.decrypt(v.wallet, keyset);
545:     if(wallet===null) return false;
546:     var privkeys = v.keys ? WalletCrypto.decrypt(v.keys, keyset) : '{}';
547:     if(privkeys===null) return false;
548:     FW.WALLET_ENCKEY = keyset;
549:     ss.setItem('wallet', wallet);
550:     FW.WALLET_KEYS = JSON.parse(privkeys);
551:     ss.removeItem('skipWalletAuth');
552:     // A successful v2 unlock confirms the vault is good, so any pre-migration
553:     // legacy blobs kept as a safety net can now be discarded.
554:     discardV1Backup();
555:     return true;

588: function migrateWalletToV2( enteredPassword, newPassword ){
589:     var wasEncrypted = parseInt(ls.getItem('walletEncrypted'))===1,
590:         storedPw     = ls.getItem('walletPassword');
591:     var v1pw;
592:     if(wasEncrypted){
593:         if(!WalletCrypto.legacyVerify(enteredPassword, storedPw)) return false;
594:         v1pw = enteredPassword;
595:     } else {
596:         v1pw = WalletCrypto.legacyConveniencePassword(storedPw);
597:     }
598:     var wallet   = WalletCrypto.decryptV1(ls.getItem('wallet'), v1pw),
599:         keysBlob = ls.getItem('walletKeys'),
600:         privkeys = keysBlob ? WalletCrypto.decryptV1(keysBlob, v1pw) : '{}';
601:     if(wallet===null || privkeys===null) return false;
602:     // The new password: the existing one for an already-encrypted wallet, or the
603:     // user's chosen password for a former convenience wallet.
604:     var newPw = wasEncrypted ? enteredPassword : newPassword;
605:     if(!newPw) return false;
606:     var mk = WalletCrypto.makeCrypto(newPw),
607:         encWallet = WalletCrypto.encrypt(wallet, mk.keyset),
608:         encKeys   = WalletCrypto.encrypt(privkeys, mk.keyset);
609:     // Verify the new blobs decrypt back to the identical plaintext BEFORE writing
610:     // anything.
611:     if(WalletCrypto.decrypt(encWallet, mk.keyset)!==wallet) return false;
612:     if(WalletCrypto.decrypt(encKeys, mk.keyset)!==privkeys) return false;
613:     // Write the v2 vault as one atomic record. The legacy v1 keys
614:     // (wallet/walletKeys/walletPassword) are left in place untouched -- they are
615:     // the natural backup, and are removed only on the next successful v2 unlock
616:     // (discardV1Backup). So a crash during migration leaves either the intact v1
617:     // wallet or a fully-written v2 vault, never a half-migrated wallet.
618:     FW.WALLET_ENCKEY = mk.keyset;
619:     ss.setItem('wallet', wallet);
620:     FW.WALLET_KEYS = JSON.parse(privkeys);
621:     writeVault(mk.descriptor);
622:     return true;
623: }

1273: function autoBtcpay(network, o){
1274:     // console.log('autoBtcpay network, o=', network, o);
1275:     var a       = ss.getItem('wallet'),
1276:         b       = ss.getItem('btcpayWallet'),
1277:         c       = false, // Flag to indicate if we should delete wallet after tx
1278:         id      = o.tx0_hash + '_' + o.tx1_hash,
1279:         fee     = FW.NETWORK_INFO.fee_info.optimal, // Use high priority fee for order matches
1280:         size    = 350,                              // 1 input is about 300 bytes... (TODO - Get actual tx size via pre-flight check)
1281:         fee_sat = getSatoshis(((fee / 1000) * size) * 0.00000001);
1282:     // Check status of the wallet and hot-swap wallet into place if needed
1283:     if(a==null){
1284:         // If no wallet/btcpayWallet is found, bail out
1285:         if(b==null){
1286:             return;
1287:         } else {
1288:             ss.setItem('wallet',b);
1289:             // Restore the stashed imported keys so imported-address orders can be
1290:             // signed while locked; HD addresses re-derive from the seed anyway.
1291:             var bk = ss.getItem('btcpayKeys');
1292:             if(bk){
1293:                 try { FW.WALLET_KEYS = JSON.parse(bk); } catch(e){ FW.WALLET_KEYS = {}; }
1294:             }
1295:             c = true;
1296:         }
1297:     }
1298:     // Handle automatically creating/signing/broadcasting the BTCpay tx
1299:     cpBtcpay(network, o.source, id, fee_sat, function(tx){
1300:         // Only proceed if we have a valid tx hash for the broadcast tx... otherwise leave in queue so we can try again
1301:         if(tx){
1302:             dialogMessage('<i class="fa fa-lg fa-check"></i> BTCPay Successful', '<center>Your BTC payment has been broadcast to the network and your order should complete shortly.' +
1303:                           '<br/><br/><a class="btn btn-success" href="' + FW.EXPLORER_API + '/tx/' + tx + '" target="_blank">View Transaction</a></center>');
1304:             // Remove the order match from the queue and check the queue again after a brief delay
1305:             removeFromBtcpayQueue(o.tx0_hash, o.tx1_hash);
1306:             setTimeout(function(){ processBtcpayQueue(); },1000);
1307:         }
1308:         // Handle removing the hot-swapped wallet + restored keys if needed, so a

1310:         if(c){
1311:             ss.removeItem('wallet');
1312:             FW.WALLET_KEYS = {};
1313:         }
1314:     });
