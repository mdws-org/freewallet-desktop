304: function initWallet(){
305:     // Presence is judged on the raw records, not on whether they parse: a
306:     // damaged vault must still be treated as a wallet, never as an invitation
307:     // to create a new one over it.
308:     var hasV2 = ls.getItem('walletVault')!==null,
309:         hasV1 = ls.getItem('wallet')!==null; // legacy layout still present (pre-migration or backup)
310:     if(hasV2 || hasV1){
311:         // A stale decrypted copy may linger in session storage from a previous
312:         // load, but the derived key is memory-only and gone after a reload, so
313:         // always start locked and require the password. Nothing auto-decrypts.
314:         ss.removeItem('wallet');
315:         FW.WALLET_KEYS = {};
316:         FW.WALLET_ENCKEY = null;
317:         // The Auto-BTCpay check runs once the unlock dialog has been answered,
318:         // so its own password dialog never opens on top of it. After an unlock
319:         // the wallet is open and the check stashes silently; after a Cancel it
320:         // is locked and the check asks (dialogEnableBtcpay), which is also
321:         // where auto-pay can be turned off. A legacy wallet cannot answer that
322:         // dialog until migration has produced a v2 vault, so the migrate
323:         // dialog runs the check only on success. The manual-payment prompt for
324:         // a queued match is held back by processBtcpayQueue itself while any
325:         // password dialog is on screen.
326:         if(hasV2){
327:             // v2 vault present (possibly with legacy backup blobs still beside
328:             // it): prompt for the password to unlock.
329:             dialogPassword(false, checkBtcpayAuth, checkBtcpayAuth);
330:         } else {
331:             // Only a legacy v1 wallet on disk: migrate to v2 on unlock.
332:             dialogMigrate(checkBtcpayAuth);
333:         }
334:     } else {
335:         dialogWelcome();
336:     }
337:     // Trigger an immediate check of if we need to update the wallet information (prices/balances)
338:     checkUpdateWallet();
339:     // Check every 60 seconds if we should update the wallet information
340:     setInterval(checkUpdateWallet, 60000);
341:     updateWalletOptions();
342:     // Populate the oracle list
343:     updateOracleList();
344:     // Populate the donation list
345:     updateDonationList();

1345: function autoBtcpay(network, o){
1346:     // console.log('autoBtcpay network, o=', network, o);
1347:     var a       = ss.getItem('wallet'),
1348:         b       = ss.getItem('btcpayWallet'),
1349:         c       = false, // Flag to indicate if we should delete wallet after tx
1350:         id      = o.tx0_hash + '_' + o.tx1_hash,
1351:         fee     = FW.NETWORK_INFO.fee_info.optimal, // Use high priority fee for order matches
1352:         size    = 350,                              // 1 input is about 300 bytes... (TODO - Get actual tx size via pre-flight check)
1353:         fee_sat = getSatoshis(((fee / 1000) * size) * 0.00000001);
1354:     // Check status of the wallet and hot-swap wallet into place if needed
1355:     if(a==null){
1356:         // If no wallet/btcpayWallet is found, bail out
1357:         if(b==null){
1358:             return;
1359:         } else {
1360:             ss.setItem('wallet',b);
1361:             // Restore the stashed imported keys so imported-address orders can be
1362:             // signed while locked; HD addresses re-derive from the seed anyway.
1363:             var bk = ss.getItem('btcpayKeys');
1364:             if(bk){
1365:                 try { FW.WALLET_KEYS = JSON.parse(bk); } catch(e){ FW.WALLET_KEYS = {}; }
1366:             }
1367:             c = true;
1368:         }
1369:     }
1370:     // Handle automatically creating/signing/broadcasting the BTCpay tx
1371:     cpBtcpay(network, o.source, id, fee_sat, function(tx){
1372:         // Only proceed if we have a valid tx hash for the broadcast tx... otherwise leave in queue so we can try again
1373:         if(tx){
1374:             dialogMessage('<i class="fa fa-lg fa-check"></i> BTCPay Successful', '<center>Your BTC payment has been broadcast to the network and your order should complete shortly.' +
1375:                           '<br/><br/><a class="btn btn-success" href="' + FW.EXPLORER_API + '/tx/' + tx + '" target="_blank">View Transaction</a></center>');
1376:             // Remove the order match from the queue and check the queue again after a brief delay
1377:             removeFromBtcpayQueue(o.tx0_hash, o.tx1_hash);
1378:             setTimeout(function(){ processBtcpayQueue(); },1000);
1379:         }
1380:         // Handle removing the hot-swapped wallet + restored keys if needed, so a
1381:         // locked session returns to holding no secret in memory after the tx.
1382:         // If the user unlocked while the tx was in flight, the session is now
1383:         // theirs (the derived key is set) and is left alone.
1384:         if(c && !FW.WALLET_ENCKEY){
1385:             ss.removeItem('wallet');
1386:             FW.WALLET_KEYS = {};
1387:         }
1388:     });
1389: }
